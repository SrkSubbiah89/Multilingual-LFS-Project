\chapter{CrewAI Agent YAML Configurations}
\label{app:crewai}

Representative YAML configurations for the four core prototype agents.
Environment secrets (API keys) are injected at runtime and not reproduced here.

\section*{B.1 --- ISCOClassifier}
\begin{lstlisting}[breaklines=true,caption={ISCOClassifier agent configuration.}]
name: isco_classifier
role: >
  Four-stage hierarchical ISCO-08 occupation coder for the LFS.
  Uses multilingual-e5-large embeddings and parent-code-filtered Qdrant
  search across four collections (major, submajor, minor, unit groups).
goal: >
  Emit a calibrated 4-digit ISCO-08 unit-group code with a full evidence
  package (retrieved text, per-stage similarities, composite confidence,
  SRE coherence), or escalate to HITL when confidence < 0.70.
llm: anthropic/claude-3-5-sonnet-20241022   # re-ranking only
temperature: 0.0                             # deterministic re-ranking
qdrant_collections:
  - isco08_major_groups      # 10 records
  - isco08_submajor_groups   # 43 records
  - isco08_minor_groups      # 130 records
  - isco08_unit_groups       # 441 records (synonym-enriched)
embeddings: intfloat/multilingual-e5-large   # 1024-dim
confidence_weights: [0.10, 0.20, 0.20, 0.50]
hitl_threshold: 0.70
skip_rerank_threshold: 0.92
keyword_prefilter: true      # anchors to major group before embedding search
\end{lstlisting}

\section*{B.2 --- SemanticRelationEngine}
\begin{lstlisting}[breaklines=true,caption={SemanticRelationEngine configuration.}]
name: semantic_relation_engine
role: >
  Deterministic three-way crosswalk between ISCO-08, ISIC Rev.~4, and
  ISCED~2011 using ILO Geneva 2012 ISCO-ISIC table and UNESCO ISCED~2011
  Operational Manual Table 7. No LLM involvement in core coherence logic.
goal: >
  Produce a SemanticCoherence score (0-1) and severity classification
  (NONE/LOW/MODERATE/HIGH). Adjust ISCO confidence by -20% to +10%.
  Trigger HITL escalation for HIGH-severity violations.
isco_isic_table: ilo_geneva_2012_isco_isic_crosswalk.json
isced_ranges_table: unesco_isced2011_table7.json
weights:
  isic: 0.55
  isced: 0.45
severity_thresholds:
  none: 0.90       # +10% confidence
  low: 0.70        # +5% confidence
  moderate: 0.50   # -5% confidence
  high: 0.0        # -20% confidence, HITL mandatory
submajor_strict_rules:
  "22": {isic_required: "Q", isced_min: 7}   # Health Professionals
  "25": {isic_preferred: "J"}                 # ICT Professionals
  "11": {isced_min: 6}                        # Chief Executives
bilingual_output: true    # explanation_en + explanation_ar in response
\end{lstlisting}

\section*{B.3 --- ConversationManager FSM}
\begin{lstlisting}[breaklines=true,caption={ConversationManager FSM configuration.}]
name: conversation_manager
role: >
  5-state FSM orchestrator for the complete Labour Force Survey
  (Sections A-K, 56 fields, ILO ICLS-19). Manages 3 employment paths
  (employed/unemployed/not-in-labour-force) with 11 skip gates.
goal: >
  Complete the LFS questionnaire accurately, enforcing skip logic
  and ILO ICLS-19 special rules (E1 wage gate, F3 re-routing) in real
  time. Coordinate ISCO, ISIC, ISCED, SRE, Validation, and PersonRegister
  agents per turn.
states: [GREETING, COLLECTING_INFO, CLARIFYING, VALIDATING, COMPLETING]
llm: ollama/llama3.2          # general tasks
temperature: 0.3
fallback_llm: claude-3-5-sonnet-20241022
field_order_fn: _get_field_order(collected_data)  # recomputed every turn
employment_paths:
  employed: [B, C_full, D, E, H, I, J, K]
  unemployed: [B, C1, F, G_if_worked, H, J, K]
  not_in_labour_force: [B, C1, G_if_worked, H, J, K]
\end{lstlisting}

\section*{B.4 --- ValidationAgent}
\begin{lstlisting}[breaklines=true,caption={ValidationAgent ILO ICLS-19 rules configuration.}]
name: validation_agent
role: >
  Real-time ILO ICLS-19 cross-answer consistency enforcer (rules R01-R10).
goal: >
  Detect all 10 ILO ICLS-19 rule violations in real time during the survey
  session and return structured ValidationResult with violated rules,
  severity, and remediation guidance.
llm: anthropic/claude-3-5-sonnet-20241022   # CRITICAL task
temperature: 0.0
rules:
  R01: Employed respondents must provide job_title and industry
  R02: Hours worked must be 0-168 per week
  R03: Unemployed (F1=yes) must provide job_search_method
  R04: University-educated (education_level in {bachelor,master,phd})
       must have ISCED >= 6
  R05: Self-employed respondents skip monthly_wage_range (E1 gate)
  R06: Platform workers (yes_primary/yes_supplementary) must name platforms
  R07: national residents must answer national_employment_program (H4 gate)
  R08: Income range must be consistent with employment_nature
  R09: Contract type required for paid_employee (D7 gate)
  R10: job_search_active=no AND available_for_work=no
       => reclassify to not_in_labour_force (F3 re-route)
\end{lstlisting}

\section*{D.1 --- Participant Information}

\noindent\textbf{Study title:}
Evaluation of a Multilingual Conversational AI System for Labour Force
Survey Administration.

\noindent\textbf{Principal investigator:}
Sivarama Krishnan Subbiah (2024MCS110007), M.Tech AI \& Data Science,
IIIT Kottayam.

\noindent\textbf{Supervisor:}
Dr.\ Goutam Mali, Department of Computer Science and Engineering, IIIT Kottayam.

\noindent\textbf{Note:}
This thesis is currently in the \textbf{pre-final phase}. The system described
is a working prototype only. No actual participant recruitment, pilot study
execution, or data collection has been conducted at this stage. The participant
information sheet and ethics checklist below represent the \textit{planned}
protocol for the forthcoming pilot study to be conducted in the final phase
(approximately 6 months from now), subject to institutional guidance.

\begin{itemize}
  \item \textbf{Purpose:} We are evaluating a new AI-assisted system for
  conducting employment surveys in English, Arabic, Urdu, Hindi, and Tagalog.

  \item \textbf{What participation involves:} You will complete a Labour
  Force Survey either through a conversational AI interface or through a
  traditional interviewer-administered survey.
  The session lasts approximately 20--30 minutes, followed by a brief
  satisfaction questionnaire.

  \item \textbf{Voluntary participation:} Participation is entirely voluntary.
  You may withdraw at any point without penalty or explanation.

  \item \textbf{Confidentiality:} All responses are pseudonymised immediately.
  Your name will not appear in any research output.

  \item \textbf{No real-world consequences:} Your responses will not be shared
  with any government agency, employer, financial institution, or immigration
  authority, and cannot influence any benefit, visa, or employment decision.

  \item \textbf{Data storage:} Pseudonymised data stored for 2 years
  post-publication, then securely deleted.
  GDPR Art.~15 right of access: you may request a copy of your
  pseudonymised responses within 30 days.
\end{itemize}
